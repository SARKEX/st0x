export function evaluateMarketOrderFill(input: MarketOrderFillInput): MarketOrderFillEvaluation;

// Pays-anchor selection rule:

//   isPartialFill = !isNoFill && actual * 10_000n < requested * MARKET_ORDER_FULL_FILL_THRESHOLD_BPS
//                                                              (= 9_970n; i.e. 99.7% threshold)

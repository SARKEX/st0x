    test.describe('TEST-09 — Limit deploy + simulated counterparty fill', () => {
      test('Sell limit deposits into OUTPUT vault, counterparty fill drains it', async ({ page, testClient, tokens, fundedAccount }) => {
        // 1. Fund tNVDA on maker (OUTPUT-vault token for Sell)
        const depositAmount = parseUnits('1', tokens.tNVDA.decimals);
        await fundErc20({
          client: testClient,
          token: tokens.tNVDA.address,
          holder: fundedAccount.address,
          amount: depositAmount,
          balanceSlot: tokens.tNVDA.balanceSlot
        });

        // 2. UI deploy the limit
        await page.goto(`${process.env.PREVIEW_URL}/trade/${tokens.tNVDA.id}`);
        await page.click('[data-testid="mode-tab"][data-mode="limit"]');
        await page.waitForSelector('[data-testid="limit-form-loaded"]'); // Pitfall 4 anchor
        await page.click('[data-testid="side-toggle"][data-side="sell"]');
        await page.fill('[data-testid="deposit-input"]', '1');           // 1 tNVDA into OUTPUT vault
        await page.fill('[data-testid="price-input"]', '999');           // arbitrary price for counterparty to take

        const beforeDeployBlock = await testClient.getBlockNumber();
        await page.click('[data-testid="deploy-submit"][data-side="sell"][data-mode="limit"]');
        await expect(page.locator('[data-testid="success-toast"]')).toBeVisible({ timeout: 30_000 });

        // 3. Read OrderAdded event from logs after deploy
        const logs = await testClient.getContractEvents({
          address: ORDERBOOK_ADDRESS,
          abi: ORDERBOOK_ABI,
          eventName: 'OrderAdded',
          fromBlock: beforeDeployBlock
        });
        expect(logs.length).toBeGreaterThanOrEqual(1);
        const order = logs[logs.length - 1].args.order;

        // 4. Assert OUTPUT vault credited the deposit (tNVDA in maker's output vault for the orderbook)
        const makerTnvdaPostDeploy = await testClient.readContract({
          address: tokens.tNVDA.address, abi: erc20Abi, functionName: 'balanceOf', args: [fundedAccount.address]
        });
        // Maker tNVDA balance dropped (deposited to orderbook escrow)
        expect(makerTnvdaPostDeploy).toBeLessThan(depositAmount);

        // 5. Counterparty fill — switch to UNFUNDED_ACCOUNT, fund with USDC, takeOrders
        const counterpartyUsdc = parseUnits('2000', tokens.USDC.decimals);
        await fundErc20({
          client: testClient,
          token: tokens.USDC.address,
          holder: UNFUNDED_ACCOUNT.address,
          amount: counterpartyUsdc,
          balanceSlot: tokens.USDC.balanceSlot
        });
        // Approve orderbook from counterparty
        await testClient.writeContract({
          account: UNFUNDED_ACCOUNT.address,
          address: tokens.USDC.address,
          abi: erc20Abi,
          functionName: 'approve',
          args: [ORDERBOOK_ADDRESS, counterpartyUsdc]
        });
        // takeOrders — fill a portion of the deployed order
        const takeOrdersArgs = [{ /* order, inputIOIndex, outputIOIndex, minimumInput, maximumInput, maximumIORatio, signedContext: [] */ /* — populated from `order` struct read above */ }];
        await testClient.writeContract({
          account: UNFUNDED_ACCOUNT.address,
          address: ORDERBOOK_ADDRESS,
          abi: ORDERBOOK_ABI,
          functionName: 'takeOrders',
          args: [{ orders: takeOrdersArgs, minimumInput: 0n, maximumInput: parseUnits('0.5', tokens.tNVDA.decimals), maximumIORatio: 2n ** 256n - 1n, data: '0x' }]
        });

        // 6. Post-fill: counterparty got tNVDA, maker's INPUT vault (USDC) credited
        const counterpartyTnvda = await testClient.readContract({
          address: tokens.tNVDA.address, abi: erc20Abi, functionName: 'balanceOf', args: [UNFUNDED_ACCOUNT.address]
        });
        const counterpartyUsdcPost = await testClient.readContract({
          address: tokens.USDC.address, abi: erc20Abi, functionName: 'balanceOf', args: [UNFUNDED_ACCOUNT.address]
        });

        expect(counterpartyUsdcPost).toBeLessThan(counterpartyUsdc);
      });

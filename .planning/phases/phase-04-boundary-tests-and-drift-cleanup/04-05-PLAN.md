```typescript
export const POST: RequestHandler = async ({ request, cookies }) => {
    const guardResponse = await requireAdmin(request, cookies, '<existing-tag>');
    if (guardResponse) return guardResponse;

    const audit = createAuditLogger(request);
    try {
        // existing handler body (UNCHANGED)
        const result = await doTheWork();
        await audit.logSuccess(
            '<EVENT_TYPE>',
            { /* relevant details: e.g., walletAddress, snapshotId — NO PII like signatures */ },
            { adminUser: 'admin' }
        );
        return json({ success: true, result });
    } catch (err) {
        // Wrap audit-log emission in its own try/catch so an audit failure
        // cannot mask the original error.
        try {
            await audit.logFailure(
                '<EVENT_TYPE>',
                { /* same details object */ },
                err instanceof Error ? err.message : String(err),
                { adminUser: 'admin' }
            );
        } catch (auditErr) {
            console.error('Audit-log emission failed:', auditErr);
        }
        // Preserve existing error-response shape (status code + body must be unchanged)
        throw err; // OR return json({ error: ... }, { status: ... }) — match existing endpoint behavior
    }
};
```

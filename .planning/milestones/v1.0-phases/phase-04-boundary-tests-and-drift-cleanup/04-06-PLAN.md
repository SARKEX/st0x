```typescript
import { vi, describe, it, expect, beforeEach, type Mock } from 'vitest';

// Mock auditLog at top-of-file scope (Pitfall 4: NOT vi.doMock)

        logFailure: vi.fn()
    }))
}));

// Mock requireAdmin to no-op (always allow) — we are testing audit, not admin gating
vi.mock('$lib/server/adminAuth', () => ({
    requireAdmin: vi.fn(async () => null)  // null means "proceed"
}));

// Mock any module the endpoint depends on for state mutation (DB, KV, cron trigger).
// Inspect the endpoint's imports and mock the underlying op so we can drive success / failure.
vi.mock('<the-endpoint-specific-dependency>', () => ({
    <opName>: vi.fn()
}));

// Import the handler AFTER mocks are established
import { POST } from '$routes/api/admin/<endpoint>/+server';
import { createAuditLogger } from '$lib/server/auditLog';

describe('admin/<endpoint> audit-log fan-out', () => {
    beforeEach(() => {
        vi.clearAllMocks();
    });

    it('logs success on happy-path POST', async () => {
        // Arrange: configure the underlying dependency to succeed
        const { <opName> } = await import('<the-endpoint-specific-dependency>');
        (<opName> as Mock).mockResolvedValueOnce({ /* successful result shape */ });

        // Act
        const event = createMockRequestEvent({
            method: 'POST',
            url: 'http://localhost/api/admin/<endpoint>',
            headers: { 'content-type': 'application/json' },
            body: JSON.stringify({ /* representative request body */ })
        });
        await POST(event);

        // Assert: createAuditLogger was called, logSuccess invocation matches expected shape
        expect(createAuditLogger).toHaveBeenCalledWith(event.request);
        const loggerInstance = (createAuditLogger as Mock).mock.results[0].value;
        expect(loggerInstance.logSuccess).toHaveBeenCalledWith(
            '<EVENT_TYPE>',                                           // e.g., 'EXCLUDED_WALLET_ADDED'
            expect.objectContaining({ /* relevant details key */ }),  // e.g., { walletAddress: ... }
            expect.objectContaining({ adminUser: 'admin' })
        );
        expect(loggerInstance.logFailure).not.toHaveBeenCalled();
    });

    it('logs failure when underlying operation throws', async () => {
        // Arrange: configure the underlying dependency to throw
        const { <opName> } = await import('<the-endpoint-specific-dependency>');
        (<opName> as Mock).mockRejectedValueOnce(new Error('underlying op failed'));

        // Act + capture
        const event = createMockRequestEvent({
            method: 'POST',
            url: 'http://localhost/api/admin/<endpoint>',
            headers: { 'content-type': 'application/json' },
            body: JSON.stringify({ /* representative request body */ })
        });
        let caught: unknown;
        try {
            await POST(event);
        } catch (e) {
            caught = e;
        }

        // Assert: logFailure called with the error message
        const loggerInstance = (createAuditLogger as Mock).mock.results[0].value;
        expect(loggerInstance.logFailure).toHaveBeenCalledWith(
            '<EVENT_TYPE>',
            expect.objectContaining({ /* relevant details key */ }),
            'underlying op failed',
            expect.objectContaining({ adminUser: 'admin' })
        );
        // Endpoint must still propagate the error (audit emission is non-blocking but does not swallow)
        expect(caught).toBeDefined();
    });
});
```

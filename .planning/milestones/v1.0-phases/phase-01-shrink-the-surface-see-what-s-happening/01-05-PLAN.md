       ```typescript
       /**
        * Server-side structured logger (pino) + request-context propagation (AsyncLocalStorage).
        *
        * Phase 1 / OBS-02 (D-07): JSON output to stdout — Vercel Logs picks it up. No external
        * drain in this phase. Required fields: request_id, wallet, route, method, status,
        * latency_ms, level, msg, error.*. Per-route log-level matrix in `pickLevelForRoute`.
        *
        * NODE-ONLY: This module imports `node:async_hooks` (AsyncLocalStorage). Vercel's Edge
        * runtime does NOT support `async_hooks` — verified at install time that no route in
        * `src/routes/api/*` exports `config = { runtime: 'edge' }` (Pitfall 2). If a future
        * route opts into Edge, that route opts out of `getLogger()` / `getRequestContext()`.
        */

       import { AsyncLocalStorage } from 'node:async_hooks';
       import { randomUUID } from 'node:crypto';
       import pino, { type Logger } from 'pino';
       import type { Handle } from '@sveltejs/kit';
       import { dev } from '$app/environment';

       interface RequestContext {
           request_id: string;
           wallet: string | null;
           route: string;
           method: string;
           start_ms: number;
       }

       const contextStore = new AsyncLocalStorage<RequestContext>();

       // Base logger — JSON stdout in prod; debug level in dev.
       const baseLogger: Logger = pino({
           level: dev ? 'debug' : 'info',
           base: {
               app: 'st0x',
               env: dev ? 'dev' : 'prod'
           },
           formatters: {
               level: (label) => ({ level: label })
           },
           redact: {
               paths: ['req.headers.authorization', 'req.headers.cookie', '*.signature', '*.privateKey'],
               censor: '[REDACTED]'
           },
           timestamp: pino.stdTimeFunctions.isoTime
       });

       /** Get the active request's logger — child carries the context fields. */
       export function getLogger(): Logger {
           const ctx = contextStore.getStore();
           if (!ctx) return baseLogger;
           return baseLogger.child({
               request_id: ctx.request_id,
               wallet: ctx.wallet,
               route: ctx.route,
               method: ctx.method
           });
       }

       /** Read the current request's context (returns undefined outside a request, e.g. at boot). */
       export function getRequestContext(): RequestContext | undefined {
           return contextStore.getStore();
       }

       /**
        * SvelteKit `Handle` hook — must be the FIRST link in the sequence chain so that:
        *  - Sentry breadcrumbs see the request_id;
        *  - bot-rejection / CSP / CORS / auth all see the same context;
        *  - the response gets `x-request-id` echoed back for client correlation.
        */
       export const requestContextHandle: Handle = async ({ event, resolve }) => {
           const request_id = event.request.headers.get('x-request-id') ?? randomUUID();
           const wallet = event.cookies.get('wallet-address') ?? null;
           const route = event.url.pathname;
           const method = event.request.method;
           const start_ms = Date.now();

           return contextStore.run(
               { request_id, wallet, route, method, start_ms },
               async () => {
                   const response = await resolve(event);
                   response.headers.set('x-request-id', request_id);

                   const status = response.status;
                   const latency_ms = Date.now() - start_ms;
                   const level = pickLevelForRoute(route, status);
                   getLogger()[level]({ status, latency_ms }, 'request');
                   return response;
               }
           );
       };

       /**
        * Per-route log level matrix (D-07).
        * status takes precedence over route to surface error / warn classes consistently.
        */
       export function pickLevelForRoute(route: string, status: number): 'info' | 'warn' | 'error' {
           if (status >= 500) return 'error';
           if (status >= 400) return 'warn';
           if (route.startsWith('/api/snapshots/')) return 'warn';
           if (route.startsWith('/api/cron/')) return 'info';
           if (route.startsWith('/api/admin/')) return 'info';
           if (route.startsWith('/api/access/')) return 'info';
           return 'info';
       }
       ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → Vercel Node serverless | Each request crosses the boundary. The request-id middleware lets us correlate logs across the entire request lifecycle (no boundary crossing for ALS itself — it's request-scoped, single-Node-instance). |
| Vercel Logs reader → admin user | Vercel Logs is admin-only-accessible (per Vercel project settings). Per D-07, pino retains full wallet — only Sentry scrubs. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-01 | Information Disclosure | pino logging Authorization headers / cookies / signatures / private keys | mitigate | Built-in pino `redact` config: paths `['req.headers.authorization', 'req.headers.cookie', '*.signature', '*.privateKey']` censored to `[REDACTED]`. Faster than beforeSend-style walks. |
| T-05-02 | Information Disclosure | URL query params with secrets (e.g., `?code=ST0X-XXXX-XXXX` on access flows, `?signature=0x...` on signature challenges) ending up in route field | accept | The route field stores `event.url.pathname` — query string is NOT included. Per RESEARCH §"Open Questions Q5" + the implementation: pathname-only logging is the simpler-and-better pattern. Verified by reading `requestContextHandle`. |
| T-05-03 | Spoofing | Client-supplied `x-request-id` header — a request that arrives with one is accepted and reused (so client + server cross-correlate) | accept | This is intentional cross-correlation. An attacker setting `x-request-id` to a target's value can poison the log stream BUT only with values they already know (no info-disclosure gain). Vercel Logs read access is admin-only. Acceptable risk for the developer-correlation benefit. RESEARCH §"Pattern 2" explicitly endorses this pattern. |
| T-05-04 | Denial of Service | AsyncLocalStorage memory leak (context store grows) | mitigate | Built-in: ALS contexts are GC'd when the request promise resolves. Pino logs are stdout, no buffer. Verified by RESEARCH §"Security Domain" line 1173. |

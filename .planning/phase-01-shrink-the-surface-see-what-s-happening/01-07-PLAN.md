     * Take-order failure capture (OBS-03 / D-08 / D-15).
     *
     * Browser-tier dual-sink:
     *  1. Sentry.captureException(err, { tags, extra: transcript }) — immediate alert + breadcrumb
     *     context. PII scrubbing handled by `beforeSend` in `src/hooks.client.ts` (Plan 01-04).
     *  2. `console.error('[take-order failed]', JSON.stringify({ts, reason, ...transcript, error}))`
     *     — long-term searchability via PostHog session replay + Vercel browser-console capture.
     *
     * Per CONTEXT D-15: this is NOT a server-relayed endpoint. marketOrderExecution.ts runs
     * client-side; pino (server-only) does NOT receive these events. If a future server-relayed
     * take-order path is introduced, OBS-03 server-tier capture would go through pino instead;
     * no such path exists today.
     *
     * Acceptance test (D-08): a dev given a single failure log entry can re-run the exact
     * subgraph quote + on-chain state read without contacting the user. The Sentry event JSON
     * OR the console-line JSON satisfies this.

    import * as Sentry from '@sentry/sveltekit';
    import type { ProcessedQuote } from '$lib/services/marketOrderExecution';

    export type TakeOrderFailureReason =
        | 'no_quotes_available'
        | 'no_walk_fills'
        | 'unhydrated_fills'
        | 'aggregated_failed'
        | 'caught_exception';

    export interface TakeOrderTranscript {
        // Quote-side state
        subgraphQuoteHash: string | null;          // sha256 hash of full quote payload, computed once
        fullQuotePayload: ProcessedQuote[];         // exact quotes the local walk used
        // On-chain read at the moment of submission
        onChainStateRead: {
            orderHash: string | null;
            vaultBalance: string | null;             // bigint serialized via .toString()
            IOIndex: { input: number | null; output: number | null };
        };
        // Derivation
        ratio: string | null;                       // hex, from quote.ratio
        slippageBps: number;
        priceCap: string | null;                    // human decimal, the value passed to SDK
        // Side semantics (per src/lib/types/orderPerspective.ts)
        side: 'bid' | 'ask';                        // counterparty side from filterQuotesForSide
        takerAction: 'Buy' | 'Sell';
        userAction: 'Buy' | 'Sell';
        mode: 'buyUpTo' | 'spendUpTo';              // anchored type
        // Identity
        walletAddress: string | null;               // Sentry's beforeSend scrubs the address
        // Cross-correlation
        request_id: string | null;                  // null on browser-only paths
        timestamp: string;                          // ISO 8601
    }

    export function captureTakeOrderFailure(
        err: unknown,
        transcript: TakeOrderTranscript,
        reason: TakeOrderFailureReason
    ): void {
        try {
            // Sink 1: Sentry — immediate alert + breadcrumb context with PII scrubbed by beforeSend
            Sentry.captureException(err, {
                tags: { failure_reason: reason, side: transcript.side },
                extra: {
                    ...transcript,
                    errorMessage: err instanceof Error ? err.message : String(err)
                }
            });
        } catch (sentryErr) {
            // Logging never throws back into caller (project convention)
            console.error('[captureTakeOrderFailure] Sentry sink failed:', sentryErr);
        }

        try {
            // Sink 2: console.error JSON line — captured by PostHog session replay + Vercel browser console
            console.error(
                '[take-order failed]',
                JSON.stringify({
                    ts: transcript.timestamp,
                    reason,
                    ...transcript,
                    error: err instanceof Error ? err.message : String(err)
                })
            );
        } catch (jsonErr) {
            console.error('[captureTakeOrderFailure] failed to serialize transcript:', jsonErr);
        }
    }
    ```

           const {
               orderSide, amount, inputMode = 'amount', slippageBps = DEFAULT_MARKET_ORDER_SLIPPAGE_BPS,
               assetToken, paymentToken, quotes, network
           } = input;

           // OBS-03 transcript — built at entry, mutated as we progress, sent on every failure path.
           // Per D-15: client-side code → Sentry + console.error JSON dual-sink. No server endpoint.
           const transcript: TakeOrderTranscript = {
               subgraphQuoteHash: null,                  // populated below from full quotes payload
               fullQuotePayload: quotes ?? [],            // initial; updated after filterQuotesForSide
               onChainStateRead: {
                   orderHash: null,
                   vaultBalance: null,
                   IOIndex: { input: null, output: null }
               },
               ratio: null,
               slippageBps,
               priceCap: null,
               side: orderSide === 'Buy' ? 'ask' : 'bid', // taker Buy crosses ask; taker Sell crosses bid
               takerAction: orderSide,
               userAction: orderSide,
               mode: inputMode === 'amount' ? 'buyUpTo' : 'spendUpTo',
               walletAddress: null,                       // populated immediately after getSignerAddress
               request_id: null,                          // browser-only path; no server context
               timestamp: new Date().toISOString()
           };

           // Helper: failWith dispatches the transcript to captureTakeOrderFailure and returns the
           // existing user-facing error object. ALL error-return paths route through this.
           const failWith = (
               reason: TakeOrderFailureReason,
               errOrMessage: unknown,
               existingError: string
           ) => {
               captureTakeOrderFailure(errOrMessage, transcript, reason);
               return { success: false, error: existingError };
           };

           try {
               const takerAddress = getSignerAddress();
               if (!takerAddress) {
                   // SKIP — not a no-liquidity scenario per RESEARCH; user disconnected wallet.
                   return { success: false, error: 'Wallet not connected. Please reconnect and try again.' };
               }
               transcript.walletAddress = takerAddress;

               // ... existing filterQuotesForSide call
               transcript.fullQuotePayload = filteredQuotes;
               // Per CONTEXT D-08 + checker W2: populate subgraphQuoteHash via SHA-256 of the
               // serialized payload so the transcript field is non-null. crypto.subtle is available
               // in browser context (marketOrderExecution.ts is client-side).
               try {
                   const hashBuf = await crypto.subtle.digest(
                       'SHA-256',
                       new TextEncoder().encode(JSON.stringify(filteredQuotes))
                   );
                   transcript.subgraphQuoteHash =
                       '0x' +
                       Array.from(new Uint8Array(hashBuf))
                           .map((b) => b.toString(16).padStart(2, '0'))
                           .join('');
               } catch {
                   // crypto.subtle should always be present in supported browsers; fall back to null
                   // rather than throwing through the trade UI.
                   transcript.subgraphQuoteHash = null;

               if (filteredQuotes.length === 0) {

               if (!worstFill?.ratio) {
                   return failWith('caught_exception', new Error('worstFill missing ratio'), 'Unable to calculate order price. Please refresh and try again.');

               transcript.ratio = worstFill.ratio;

               // ... existing emergencyRatioHex computation
               if (!emergencyRatioHex) {
                   return failWith('caught_exception', new Error('emergencyRatioHex null'), 'Unable to calculate order price. Please refresh and try again.');
               }
               transcript.priceCap = humanPriceCapStr; // or whatever the existing variable name is

               // ... existing aggregated path (firstQuote check)
               if (!firstQuote.orderData || !firstQuote.sgOrder) {

               transcript.onChainStateRead.IOIndex.input = firstQuote.inputIOIndex ?? null;
               transcript.onChainStateRead.IOIndex.output = firstQuote.outputIOIndex ?? null;

                   return failWith('aggregated_failed', new Error('indexedFills empty in fallback'), 'Unable to prepare order transaction. Please retry.');
               }

                   return failWith('aggregated_failed', new Error(txError ?? 'Transaction failed'), txError || 'Order failed');
               }

               // success path unchanged
               return { success: true, /* ... */ };
           } catch (error) {
               return failWith('caught_exception', error, error instanceof Error ? error.message : 'Order failed');
           }
       }
       ```
